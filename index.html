document.getElementById("cadastro").addEventListener("submit", async function(e) { e.preventDefault(); const nome = document.getElementById("nome").value.trim(); const nick = document.getElementById("nick").value.trim(); const resultado = document.getElementById("resultado"); const botao = this.querySelector("button"); if (!nome || !nick) { resultado.textContent = "Preencha todos os campos."; return; } botao.disabled = true; botao.textContent = "ENVIANDO..."; try { const resposta = await fetch( "https://sorteio-robux-avila.rafa766936.workers.dev/", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ nome, nick }) } ); const dados = await resposta.json(); if (!resposta.ok || !dados.sucesso) { throw new Error(dados.erro || "Falha no cadastro"); } resultado.textContent = "Cadastro enviado com sucesso!"; this.reset(); } catch (erro) { resultado.textContent = "Não foi possível enviar. Tente novamente."; } finally { botao.disabled = false; botao.textContent = "PARTICIPAR DO SORTEIO"; } });
